def solve_challenge(challenge_text):
    """Extrai e resolve a equação do AI Verification Challenge do Moltbook."""
    text = challenge_text.lower()
    
    words_to_num = {
        "zero": 0, "one": 1, "two": 2, "three": 3, "four": 4,
        "five": 5, "six": 6, "seven": 7, "eight": 8, "nine": 9,
        "ten": 10, "twenty": 20, "thirty": 30, "forty": 40, "fifty": 50
    }
    
    numbers = []
    for word, val in words_to_num.items():
        if word in text:
            numbers.append(val)
            
    digits = re.findall(r'\b\d+\b', text)
    for d in digits:
        numbers.append(float(d))

    if len(numbers) < 2:
        print(f"[Desafio] Não foi possível extrair números de: '{challenge_text}'")
        return None

    num1, num2 = numbers[0], numbers[1]
    
    if "momentum" in text or "*" in text or "times" in text or "multiplied" in text:
        result = num1 * num2
    elif "slows" in text or "-" in text or "minus" in text:
        result = num1 - num2
    elif "adds" in text or "+" in text or "plus" in text:
        result = num1 + num2
    elif "divided" in text or "/" in text:
        result = num1 / num2 if num2 != 0 else 0.0
    else:
        result = num1 * num2

    ans = f"{result:.2f}"
    print(f"[Desafio] Calculado com sucesso: {ans}")
    return ans

def verify_content(verification_data):
    """Envia a solução do desafio matemático para publicação."""
    code = verification_data.get("verification_code")
    text = verification_data.get("challenge_text")
    if code and text:
        ans = solve_challenge(text)
        if ans:
            res = requests.post(f"{BASE_URL}/verify", headers=HEADERS, json={"verification_code": code, "answer": ans})
            if res.status_code == 200:
                print("[Verificação] Conteúdo aprovado e publicado na rede!")
            else:
                print(f"[Verificação] Falha na resposta: {res.text}")

def heartbeat_routine():
    """Rotina executada periodicamente para manter o agente ativo."""
    print(f"\n==================================================")
    print(f"[{time.strftime('%Y-%m-%d %H:%M:%S')}] Iniciando checagem do EscapularioSanctitas...")
    
    try:
        # 1. Consultar o painel principal (/home)
        res = requests.get(f"{BASE_URL}/home", headers=HEADERS)
        if res.status_code != 200:
            print(f"[Erro] Falha ao acessar /home: {res.status_code}")
            return
            
        home_data = res.json()
        account = home_data.get("your_account", {})
        print(f"Status da Conta -> Karma: {account.get('karma')} | Notificações: {account.get('unread_notification_count')}")

        # 2. Responder a interações nos próprios posts
        activity = home_data.get("activity_on_your_posts", [])
        for item in activity:
            post_id = item.get("post_id")
            print(f"Novos comentários detectados no post '{item.get('post_title')}'.Lendo conversas...")
            
            # Buscar comentários recentes
            comments_res = requests.get(f"{BASE_URL}/posts/{post_id}/comments?sort=new&limit=5", headers=HEADERS)
            if comments_res.status_code == 200:
                comments = comments_res.json().get("comments", [])
                if comments:
                    last_comment = comments[0]
                    # Enviar réplica cortês e profunda
                    reply_payload = {
                        "content": "Agradeco pela reflexao compartilhada. A busca sincera pela verdade exige justamente essa disposicao ao dialogo intelectual e a clareza de principios.",
                        "parent_id": last_comment.get("id")
                    }
                    reply_res = requests.post(f"{BASE_URL}/posts/{post_id}/comments", headers=HEADERS, json=reply_payload)
                    if reply_res.status_code == 200:
                        r_data = reply_res.json()
                        if r_data.get("post", {}).get("verification"):
                            verify_content(r_data["post"]["verification"])

        # 3. Ler o Feed e fazer Upvote em discussões relevantes
        feed_res = requests.get(f"{BASE_URL}/feed?sort=hot&limit=5", headers=HEADERS)
        if feed_res.status_code == 200:
            posts = feed_res.json().get("posts", [])
            for p in posts:
                # Dar upvote para incentivar boa conversação
                requests.post(f"{BASE_URL}/posts/{p.get('id')}/upvote", headers=HEADERS)

        # 4. Decisão de criar um novo post (sorteio aleatório para respeitar rate limits)
        if random.random() < 0.3:  # 30% de chance por ciclo
            topic = random.choice(TOPICOS_REFLEXAO)
            print(f"Gerando nova reflexão no submolt 'm/{topic['submolt']}'...")
            post_payload = {
                "submolt_name": topic["submolt"],
                "title": topic["title"],
                "content": topic["content"]
            }
            post_res = requests.post(f"{BASE_URL}/posts", headers=HEADERS, json=post_payload)
            if post_res.status_code == 200:
                p_data = post_res.json()
                if p_data.get("post", {}).get("verification"):
                    verify_content(p_data["post"]["verification"])

    except Exception as e:
        print(f"[Exceção] Ocorreu um erro no ciclo: {e}")

def run_24_7():
    print("Iniciando o agente autônomo EscapularioSanctitas [Modo 24/7]...")
    while True:
        heartbeat_routine()
        # Aguarda 30 minutos (1800 segundos) entre cada ciclo
        time.sleep(1800)

if __name__ == "__main__":
    run_24_7()
